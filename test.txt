function generateDailyLog(sheet, startDate) {
  const logStartRow = 4;

  const colDate = 25;       // Y
  const colDaily = 26;      // Z (дневное кол-во ✅)
  const colCumulative = 27; // AA (накопительно)
  const colText = 28;       // AB ("N задач")

  // Очистка старого лога (Y:AC, 365 строк)
  sheet.getRange(logStartRow, colDate, 365, 5).clearContent();

  // Заголовки
  sheet.getRange("Y2").setValue("📅 ДНЕВНИК")
    .setFontWeight("bold")
    .setFontColor(CONFIG.colors.textMain);

  sheet.getRange("Y3").setValue("Дата").setFontSize(8).setFontColor(CONFIG.colors.textMuted);
  sheet.getRange("Z3").setValue("✅").setFontSize(8).setFontColor(CONFIG.colors.textMuted);
  sheet.getRange("AA3").setValue("Σ").setFontSize(8).setFontColor(CONFIG.colors.textMuted); // накопительно
  sheet.getRange("AB3").setValue("Результат").setFontSize(8).setFontColor(CONFIG.colors.textMuted);

  const weekBlocks = detectWeekBlocks(sheet);
  const colLetters = ["C", "D", "E", "F", "G", "H", "I"]; // ПН-ВС

  let currentRow = logStartRow;

  weekBlocks.forEach((block, index) => {
    let weekStart = new Date(startDate);
    weekStart.setDate(startDate.getDate() + (index * 7));

    const rStart = block.headerRow + 1;
    const rEnd = block.habitsEnd;

    for (let d = 0; d < 7; d++) {
      const dayDate = new Date(weekStart);
      dayDate.setDate(weekStart.getDate() + d);

      // Y: дата
      sheet.getRange(currentRow, colDate)
        .setValue(dayDate)
        .setNumberFormat("dd.MM ddd");

      // Z: дневной COUNTIF ✅
      const colLetter = colLetters[d];
      const formulaDaily = `=COUNTIF(${colLetter}${rStart}:${colLetter}${rEnd}, "✅")`;
      sheet.getRange(currentRow, colDaily)
        .setFormula(formulaDaily)
        .setHorizontalAlignment("center");

      // AA: накопительно (с начала лога)
      // Если Z пустая (на всякий), то AA пустая
      const formulaCum = `=IF(Z${currentRow}="",,SUM($Z$${logStartRow}:Z${currentRow}))`;
      sheet.getRange(currentRow, colCumulative)
        .setFormula(formulaCum)
        .setHorizontalAlignment("center");

      // AB: текст "N задач"
      const formulaText = `=Z${currentRow} & " задач"`;
      sheet.getRange(currentRow, colText)
        .setFormula(formulaText)
        .setFontColor(CONFIG.colors.textMuted)
        .setFontSize(9);

      currentRow++;
    }

    // Пустая строка между неделями — только если это НЕ последняя неделя
    if (index < weekBlocks.length - 1) currentRow++;
  });
}
