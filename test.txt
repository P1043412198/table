function onEdit(e) {
  try {
    if (!e || !e.range) return;
    const range = e.range;
    const sheet = range.getSheet();
    if (sheet.getName() !== CONFIG.sheetName) return;

    const col = range.getColumn();
    const row = range.getRow();
    if (col < 3 || col > 9) return;     // Только колонки статусов по дням
    if (row < 14) return;                // Пропускаем шапку/декор

    const val = e.value;
    const prev = e.oldValue;
  } catch (err) {
    console.error(err);
  }
}
function generateDailyLog(sheet, startDate) {
  const logStartRow = 4;
  const colDate = 25; // Y
  const colCount = 26; // Z
  const colEff = 27;  // AA (для красоты или доп. данных)

  // Очистка старого лога
  sheet.getRange(logStartRow, colDate, 365, 5).clearContent();
  
  // Заголовки лога
  sheet.getRange("Y2").setValue("📅 ДНЕВНИК").setFontWeight("bold").setFontColor(CONFIG.colors.textMain);
  sheet.getRange("Y3").setValue("Дата").setFontSize(8).setFontColor(CONFIG.colors.textMuted);
  sheet.getRange("Z3").setValue("✅").setFontSize(8).setFontColor(CONFIG.colors.textMuted);
  sheet.getRange("AA3").setValue("Результат").setFontSize(8).setFontColor(CONFIG.colors.textMuted);

  // Получаем координаты блоков недель
  const weekBlocks = detectWeekBlocks(sheet);
  const colLetters = ["C", "D", "E", "F", "G", "H", "I"]; // ПН-ВС
  
  let currentRow = logStartRow;

  weekBlocks.forEach((block, index) => {
    let weekStart = new Date(startDate);
    weekStart.setDate(startDate.getDate() + (index * 7));

    // Определяем диапазон привычек (от заголовка дней до строки KPI)
    // block.headerRow - это строка с "ПН, ВТ...", привычки начинаются сразу за ней
    // block.habitsEnd - это строка перед KPI
    const rStart = block.headerRow + 1;
    const rEnd = block.habitsEnd;

    for (let d = 0; d < 7; d++) {
      let dayDate = new Date(weekStart);
      dayDate.setDate(weekStart.getDate() + d);
      
      // 1. Дата (Столбец Y)
      const cellDate = sheet.getRange(currentRow, colDate);
      cellDate.setValue(dayDate).setNumberFormat("dd.MM ddd"); // Пример: 20.01 Пн

      // 2. Формула подсчета (Столбец Z)
      // Считаем галочки в конкретном столбце дня
      const colLetter = colLetters[d];
      // Формула: =COUNTIF(C15:C20; "✅")
      const formulaCount = `=COUNTIF(${colLetter}${rStart}:${colLetter}${rEnd}, "✅")`;
      
      const cellCount = sheet.getRange(currentRow, colCount);
      cellCount.setFormula(formulaCount).setHorizontalAlignment("center");

      // 3. Текстовый результат (Столбец AA - то, что вы просили)
      // Формула склеивает текст: "5 задач"
      const formulaText = `=Z${currentRow} & " задач"`;
      sheet.getRange(currentRow, colEff).setFormula(formulaText)
           .setFontColor(CONFIG.colors.textMuted).setFontSize(9);

      currentRow++;
    }
    // Добавляем пустую строку между неделями в логе для красоты (опционально)
    currentRow++; 
  });
}
function getCycleStartDate(rawDate) {
  const base = new Date(rawDate);
  if (!(base instanceof Date) || isNaN(base)) return null;
  normalizeToStartOfDay(base);
  const day = base.getDay();
  const diff = base.getDate() - day + (day === 0 ? -6 : 1);
  base.setDate(diff);
  return base;
}

function normalizeToStartOfDay(date) {
  date.setHours(0, 0, 0, 0);
  return date;
}

function applyGlobalColors(sheet) {
   const range = sheet.getRange("C14:I1000");
   const r1 = SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('✅').setBackground(CONFIG.colors.status.doneBg).setFontColor(CONFIG.colors.status.doneText).setRanges([range]).build();
   const r2 = SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('❌').setBackground(CONFIG.colors.status.failBg).setFontColor(CONFIG.colors.status.failText).setRanges([range]).build();
   const r3 = SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('🟡').setBackground(CONFIG.colors.status.planBg).setFontColor(CONFIG.colors.status.planText).setRanges([range]).build();
   const r4 = SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('💤').setBackground(CONFIG.colors.status.skipBg).setFontColor(CONFIG.colors.status.skipText).setRanges([range]).build();
   sheet.setConditionalFormatRules([r1, r2, r3, r4]);
}

function promptWeeksCount(ui, currentValue) {
  const prompt = ui.prompt(
    '⏳ Длительность цикла',
    'Введите количество недель (2–8). Пусто = оставить ' + currentValue + ' недель.',
    ui.ButtonSet.OK_CANCEL
  );

  if (prompt.getSelectedButton() === ui.Button.CANCEL) return currentValue;

  const raw = parseInt(prompt.getResponseText(), 10);
  if (isNaN(raw)) return currentValue;

  return Math.min(8, Math.max(2, raw));
}

function columnLetter(colNumber) {
  let temp = colNumber;
  let letter = '';
  while (temp > 0) {
    const mod = (temp - 1) % 26;
    letter = String.fromCharCode(65 + mod) + letter;
    temp = Math.floor((temp - mod) / 26);
  }
  return letter;
}

function detectLogDays(logSheet, startCol) {
  const lastRow = logSheet.getLastRow();
  if (lastRow < 4) return 0;

  const dates = logSheet.getRange(4, startCol, lastRow - 3, 1).getValues();
  let count = 0;
  dates.forEach(row => {
    const val = row[0];
    if (val instanceof Date && !isNaN(val)) count++;
  });
  return count;
}

function detectWeekBlocks(sheet) {
  const weekLabels = sheet.createTextFinder("НЕДЕЛЯ").findAll() || [];
  const kpiLabels = sheet.createTextFinder(ANCHOR_KPI).findAll() || [];

  const weekStarts = weekLabels
    .filter(r => r.getColumn() === 2)
    .map(r => r.getRow())
    .sort((a, b) => a - b);

  const kpiRows = kpiLabels
    .filter(r => r.getColumn() === 2)
    .map(r => r.getRow())
    .sort((a, b) => a - b);

  const blocks = [];
  for (let i = 0; i < weekStarts.length; i++) {
    const startRow = weekStarts[i];
    const headerRow = startRow + 2;
    const habitsStart = headerRow + 1;
    const kpiRow = kpiRows[i] || (habitsStart + 1);
    const habitsEnd = Math.max(habitsStart, kpiRow - 1);
    blocks.push({ startRow, headerRow, habitsStart, habitsEnd });
  }
  return blocks;
}

function clearLogCharts(sheet, startCol) {
  const titlesToRemove = ['Итоговая эффективность (накопительно)', 'Дневной баланс (+/-)', 'Эффективность месяца'];
  sheet.getCharts().forEach(chart => {
    const opts = chart.getOptions() || {};
    const title = (opts.title || '').toString();
    const ranges = chart.getRanges ? chart.getRanges() : [];

    const overlapsLog = ranges.some(r => {
      const cStart = r.getColumn();
      const cEnd = cStart + r.getNumColumns() - 1;
      return cStart <= startCol + 3 && cEnd >= startCol;
    });

    const matchTitle = titlesToRemove.indexOf(title) !== -1 || /эффективность месяца/i.test(title);
    if (matchTitle || overlapsLog) {
      sheet.removeChart(chart);
    }
  });

  SpreadsheetApp.flush();
}

 */
function removeOverallPieCharts(sheet) {
  sheet.getCharts().forEach(chart => {
    const ranges = chart.getRanges ? chart.getRanges() : [];
    const touchesPieData = ranges.some(r => r.getColumn() === 21 && r.getRow() === 1 && r.getNumColumns() >= 2 && r.getNumRows() >= 5);
    const touchesPieCard = chart.getContainerInfo && (() => {
      const info = chart.getContainerInfo();
      if (!info) return false;
      const anchorRow = info.getAnchorRow();
      const anchorCol = info.getAnchorColumn();
      return anchorRow >= 11 && anchorRow <= 16 && anchorCol >= 2 && anchorCol <= 6;
    })();

    if (touchesPieData || touchesPieCard) {
      sheet.removeChart(chart);
    }
  });
}

function buildOverallPie(sheet) {
  // 1. Подготовка данных (без изменений)
  const donutDataRange = sheet.getRange("U1:V5");
  donutDataRange.clearContent();

  const totalDone = `COUNTIF(C14:I1000, "✅")`;

  sheet.getRange("U1").setValue("Category");
  sheet.getRange("V1").setValue("Value");

  // Важно: порядок цветов зависит от порядка строк здесь
  sheet.getRange("U2").setValue("Done"); 
  sheet.getRange("V2").setFormula(`=${totalDone}`); // Зеленый

  sheet.getRange("U3").setValue("Fail");
  sheet.getRange("V3").setFormula(`=COUNTIF(C14:I1000, "❌")`); // Красный

  sheet.getRange("U4").setValue("Plan");
  sheet.getRange("V4").setFormula(`=COUNTIF(C14:I1000, "🟡")`); // Оранжевый

  sheet.getRange("U5").setValue("Skip");
  sheet.getRange("V5").setFormula(`=COUNTIF(C14:I1000, "💤")`); // Серый

  // Убеждаемся, что фон карточки белый
  sheet.getRange("B11:F18").setBackground(CONFIG.colors.card);

  SpreadsheetApp.flush();

  // 2. Настройка внешнего вида и позиции
  const chartBuilder = sheet.newChart()
    .asPieChart()
    .addRange(sheet.getRange("U1:V5"))
    .setNumHeaders(1)
    
    // --- ГЛАВНАЯ НАСТРОЙКА ПОЗИЦИИ ---
    // 11 = Строка (сразу под цифрой %)
    // 2  = Колонка B (начало карточки)
    // 135 = Сдвиг вправо (в пикселях), чтобы попасть в центр
    // 10  = Сдвиг вниз (в пикселях), небольшой отступ
    .setPosition(11, 2, 135, 10) 
    
    .setOption('legend', { position: 'none' })
    .setOption('pieSliceText', 'none')
    
    // Превращаем пирог в "Пончик" (как на скрине)
    .setOption('pieHole', 0.6) 
    
    // Убираем рамку самой диаграммы, чтобы она слилась с карточкой
    .setOption('backgroundColor', { fill: 'transparent' }) 
    
    .setOption('colors', [
       CONFIG.colors.status.doneText, // Цвет для Done
       CONFIG.colors.status.failText, // Цвет для Fail
       CONFIG.colors.status.planText, // Цвет для Plan
       CONFIG.colors.status.skipText  // Цвет для Skip (используем серый текст, чтобы было видно)

    .setOption('height', 130)
    // Растягиваем круг внутри виджета на максимум
    .setOption('chartArea', { left: '5%', top: '5%', width: '90%', height: '90%' });

  sheet.insertChart(chartBuilder.build());

    .setOption('pointSize', 0)
    
    // --- ГЛАДКОСТЬ ЛИНИИ ---
    .setOption('curveType', 'function') 
    // -----------------------
    
    // ОСИ
    .setOption('hAxis', { 
      textPosition: 'none', 
      gridlines: { count: 0 },
      baselineColor: 'none'
    })
    .setOption('vAxis', {

function createDashboard(sheet) {

  sheet.getRange("B2:O2").merge().setValue("ТРЕКЕР ПРИВЫЧЕК")
       .setFontSize(36).setFontWeight("bold").setFontColor(CONFIG.colors.accent)
       .setHorizontalAlignment("left").setVerticalAlignment("middle")
       .setFontFamily(CONFIG.fontHeader);

  // --- ЛЕВЫЙ БЛОК (ОБЩИЙ ПРОЦЕНТ) ---
  const leftBox = sheet.getRange("B5:F18");
  leftBox.setBackground(CONFIG.colors.card)
         .setBorder(true, true, true, true, null, null, CONFIG.shadow, SpreadsheetApp.BorderStyle.SOLID);
  
  sheet.getRange("B5:F5").merge().setValue("ОБЩИЙ ПРОГРЕСС").setFontSize(9).setFontWeight("bold").setHorizontalAlignment("center").setFontColor(CONFIG.colors.textMuted).setVerticalAlignment("bottom");
  
  const totalCheck = `COUNTIF(C14:I1000, "✅")`;
  const totalAll = `(COUNTIF(C14:I1000, "✅")+COUNTIF(C14:I1000, "❌")+COUNTIF(C14:I1000, "🟡"))`;
  const percentMath = `IF(${totalAll}=0, 0, ${totalCheck}/${totalAll})`;

  // Большая цифра
  sheet.getRange("B6:F9").merge().setFormula("=" + percentMath)
       .setNumberFormat("0%").setFontSize(55).setFontWeight("bold") 
       .setHorizontalAlignment("center").setVerticalAlignment("middle")
       .setFontFamily(CONFIG.fontMono)
       .setFontColor(CONFIG.colors.textMain);
  
  // ASCII Bar
  const blocksTotal = 15;
  const filledBlocks = `INT(${percentMath} * ${blocksTotal})`;
  const emptyBlocks = `${blocksTotal} - ${filledBlocks}`;
  const asciiBar = `REPT("▮", ${filledBlocks}) & REPT("▯", ${emptyBlocks})`;

  sheet.getRange("B10:F10").merge().setFormula("=" + asciiBar)
       .setFontFamily("Courier New")
       .setHorizontalAlignment("center").setFontSize(12).setFontColor(CONFIG.colors.accent);

  // --- ПРАВЫЙ БЛОК (ЦИФРЫ) ---
  const rightBox = sheet.getRange("G5:O11");
  rightBox.setBackground(CONFIG.colors.card)
         .setBorder(true, true, true, true, null, null, CONFIG.shadow, SpreadsheetApp.BorderStyle.SOLID);
  
  // Заголовки (сдвинуты влево)
  const labels = [
    {r: "G5:H5", t: "ГОТОВО"}, {r: "I5:J5", t: "ПЛАН"}, 
    {r: "K5:L5", t: "СРЫВ"}
  ];
  labels.forEach(l => {
    sheet.getRange(l.r).merge().setValue(l.t).setHorizontalAlignment("center").setVerticalAlignment("bottom")
         .setFontSize(9).setFontWeight("bold").setFontColor(CONFIG.colors.textMuted);
  });
  
  // Метка для "Отдых/Пропуск"
  sheet.getRange("M5:M5").setValue("ОТДЫХ").setHorizontalAlignment("center").setVerticalAlignment("bottom")
       .setFontSize(9).setFontWeight("bold").setFontColor(CONFIG.colors.textMuted);

  // Цифры
  sheet.getRange("G6:H9").merge().setFormula(`=${totalCheck}`).setHorizontalAlignment("center").setVerticalAlignment("middle")

/**
 * ➕ ДОБАВИТЬ ПРИВЫЧКУ
 */
function addHabitToAllWeeks(opts) {
  const options = opts || {};
  const defaultStatus = options.defaultStatus;

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.sheetName);
  if (!sheet) return;
  
  const textFinder = sheet.createTextFinder(ANCHOR_KPI);
  const ranges = textFinder.findAll();
  if (ranges.length === 0) return;
  
  let rows = ranges.map(r => r.getRow());
  rows.sort((a, b) => a - b);

  const firstKpiRow = rows[0];
  let tempRow = firstKpiRow - 1;
  let habitCount = 0;
  while (habitCount < 20) {
    const val = sheet.getRange(tempRow, 3).getValue();
    if (val === "ПН") break; 
    tempRow--;
    habitCount++;
  }
  
  if (habitCount >= CONFIG.limits.habits) {
    SpreadsheetApp.getUi().alert(`Лимит: максимум ${CONFIG.limits.habits} привычек.`);
    return;
  }

  let previousHabitRow = null; 
  let rowsAddedOffset = 0;     

  for (let i = 0; i < rows.length; i++) {
    const kpiRow = rows[i] + rowsAddedOffset;
    
    sheet.insertRowBefore(kpiRow);
    const newRow = kpiRow;
    
    sheet.getRange(newRow, 1, 1, 20).setBackground(CONFIG.colors.bg).setFontWeight("normal").setBorder(null, null, null, null, null, null);
    sheet.getRange(newRow, 2, 1, 14).setBackground(CONFIG.colors.card);
    
    const nameCell = sheet.getRange(newRow, 2);
    nameCell.setFontSize(9).setFontColor(CONFIG.colors.textMain).setVerticalAlignment("middle").setHorizontalAlignment("left");
    
    if (i === 0) {
      nameCell.setValue(`Привычка ${habitCount + 1}`);
      previousHabitRow = newRow;
    } else {
      nameCell.setFormula(`=B${previousHabitRow}`);
      previousHabitRow = newRow;
    }

    const checkRange = sheet.getRange(newRow, 3, 1, 7);
    const rule = SpreadsheetApp.newDataValidation().requireValueInList(STATUS_SYMBOLS, true).build();
    checkRange.setDataValidation(rule).setHorizontalAlignment("center").setVerticalAlignment("middle");
    if (defaultStatus && STATUS_SYMBOLS.indexOf(defaultStatus) !== -1) {
      checkRange.setValues([Array(7).fill(defaultStatus)]);
    }

    // Огоньки (Стрик)
    const rangeDays = `C${newRow}:I${newRow}`;
    const fireFormula = `=IF(COUNTIF(${rangeDays}, "✅")=7, "⚡⚡", IF(COUNTIF(${rangeDays}, "✅")>=5, "⚡", SPARKLINE(COUNTIF(${rangeDays}, "✅"), {"charttype","bar";"max",7;"color1","${CONFIG.colors.textMain}";"color2","${CONFIG.colors.bg}"})))`;
    sheet.getRange(newRow, 10).setFormula(fireFormula).setVerticalAlignment("middle").setFontSize(8).setFontColor(CONFIG.colors.accent);

    sheet.setRowHeight(newRow, 24); 

    // Обновление статистики
    const actualKpiRow = newRow + 1;
    
    let headerRow = newRow - 1;
    let safeguard = 0;
    while(sheet.getRange(headerRow, 3).getValue() !== "ПН" && safeguard < 20) {
      headerRow--;
      safeguard++;
    }
    const habitsStartRow = headerRow + 1;
    const habitsEndRow = newRow; 
    
    const rangeWeekAll = `C${habitsStartRow}:I${habitsEndRow}`;
    
    const colLetters = ["C", "D", "E", "F", "G", "H", "I"];
    for(let d=0; d<7; d++) {
      const col = colLetters[d];
      const dayRange = `${col}${habitsStartRow}:${col}${habitsEndRow}`;
      const kpiForm = `=IF((COUNTIF(${dayRange},"✅")+COUNTIF(${dayRange},"❌")+COUNTIF(${dayRange},"🟡"))=0, 0, COUNTIF(${dayRange},"✅")/(COUNTIF(${dayRange},"✅")+COUNTIF(${dayRange},"❌")+COUNTIF(${dayRange},"🟡")))`;
      sheet.getRange(actualKpiRow, 3 + d).setFormula(kpiForm).setNumberFormat("0%");
    }
    
    let statLabelRow = actualKpiRow + 1;
    safeguard = 0;
    while(sheet.getRange(statLabelRow, 12).getValue() !== ANCHOR_GRADE && safeguard < 15) {
      statLabelRow++;
      safeguard++;
    }
    
    if (sheet.getRange(statLabelRow, 12).getValue() === ANCHOR_GRADE) {
        const percentRow = statLabelRow - 3; 
        const barRow = statLabelRow - 1;     
        const columnsRow = statLabelRow + 1; 
        
        const weekPercentFormula = `IF((COUNTIF(${rangeWeekAll}, "✅")+COUNTIF(${rangeWeekAll}, "❌")+COUNTIF(${rangeWeekAll}, "🟡"))=0, 0, COUNTIF(${rangeWeekAll}, "✅")/(COUNTIF(${rangeWeekAll}, "✅")+COUNTIF(${rangeWeekAll}, "❌")+COUNTIF(${rangeWeekAll}, "🟡")))`;
        
        sheet.getRange(percentRow, 12, 2, 4).merge()
             .setFormula(`=${weekPercentFormula}`)
             .setNumberFormat("0%")
             .setFontSize(36).setFontWeight("bold").setHorizontalAlignment("center").setVerticalAlignment("middle").setFontFamily(CONFIG.fontMono);

        const barFormula = `=SPARKLINE(${weekPercentFormula}, {"charttype", "bar"; "max", 1; "color1", "${CONFIG.colors.textMain}"; "color2", "${CONFIG.colors.progressBarEmpty}"})`;
        sheet.getRange(barRow, 12, 1, 4).merge().setFormula(barFormula);

        const kpiDailyRange = `C${actualKpiRow}:I${actualKpiRow}`;
        const columnFormula = `=SPARKLINE(${kpiDailyRange}, {"charttype", "column"; "color", "${CONFIG.colors.accent}"; "highcolor", "${CONFIG.colors.status.doneText}"; "ymin", 0; "ymax", 1})`;
        
        sheet.getRange(columnsRow, 12, 1, 4).merge()
             .setFormula(columnFormula)
             .setBackground(CONFIG.colors.card);
    }

    rowsAddedOffset++; 
  }
  
  SpreadsheetApp.getActive().toast("Привычка добавлена!", "ОК");
}

function removeHabitFromAllWeeks() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.sheetName);
  if (!sheet) return;
  const ui = SpreadsheetApp.getUi();
  const textFinder = sheet.createTextFinder(ANCHOR_KPI);
  const ranges = textFinder.findAll();
  if (ranges.length === 0) return;
  
  const checkRow = ranges[0].getRow();
  
  if (sheet.getRange(checkRow - 1, 3).getValue() === "ПН") {
     ui.alert("Нет привычек для удаления!"); return;
  }
  
  const response = ui.alert('Удалить последнюю привычку во всех неделях?', ui.ButtonSet.YES_NO);
  if (response == ui.Button.NO) return;
  
  ranges.sort((a, b) => b.getRow() - a.getRow());
  
  ranges.forEach(range => {
    sheet.deleteRow(range.getRow() - 1);
  });
  
  SpreadsheetApp.getActive().toast("Удалено.", "ОК");
}

function addGlobalGoal() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(CONFIG.sheetName);
  if (!sheet) return;

  const finder = sheet.createTextFinder(ANCHOR_GOALS).findNext();
  if (!finder) return;
  
  const startRow = finder.getRow();
  
  let count = 0;
  let lastRow = startRow;
  
  while (count < 20) {
    if (sheet.getRange(lastRow + 1, 17).getBackground() !== CONFIG.colors.card) break;
    lastRow++;
    count++;
  }
  
  if (count >= CONFIG.limits.goals) return;
  
  const insertRow = lastRow + 1;
  sheet.getRange(insertRow, 17, 1, 4).insertCells(SpreadsheetApp.Dimension.ROWS);
  
  const r = insertRow;
  sheet.getRange(r, 17, 1, 4).setBackground(CONFIG.colors.card);
  sheet.getRange(r, 17).insertCheckboxes();
  sheet.getRange(r, 18, 1, 3).merge().setValue("Новая цель")
       .setFontSize(9).setVerticalAlignment("middle").setFontColor(CONFIG.colors.textMain);
       
  const rule = SpreadsheetApp.newConditionalFormatRule()
      .whenFormulaSatisfied(`=$Q${r}=TRUE`).setFontColor(CONFIG.colors.textMuted).setStrikethrough(true)
      .setRanges([sheet.getRange(r, 18, 1, 3)]).build();
  let rules = sheet.getConditionalFormatRules();
  rules.push(rule);
  sheet.setConditionalFormatRules(rules);
}
